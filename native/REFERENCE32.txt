local reference environment, not a relocatable redistributable toolchain.
Wrapper:
#!/bin/sh
root=$(CDPATH= cd -- "$(dirname -- "$0")/root" && pwd)
exec gcc-15 -m32 --sysroot="$root" -B"$root/usr/lib/gcc/x86_64-linux-gnu/15/32/" -B"$root/usr/lib32/" -isystem "$root/usr/include" -isystem /usr/include/x86_64-linux-gnu -isystem /usr/include -L"$root/usr/lib32" -Wl,--dynamic-linker="$root/usr/lib32/ld-linux.so.2" -Wl,-rpath="$root/usr/lib32" "$@"
